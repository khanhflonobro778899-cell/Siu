-- Hàm request đa năng cho mọi loại Executor
local function KhongKichRequest(options)
    local req = (syn and syn.request) or (http and http.request) or http_request or (fluxus and fluxus.request) or request
    if req then
        return req(options)
    end
    return nil
end

-- Hàm đóng gói và bắn dữ liệu về Web trung tâm
local function sendTrackerData(web_url)
    if not game.JobId or game.JobId == "" then 
        warn("⚠️ [Tracker] Không tìm thấy JobId của Server!")
        return 
    end
    
    local data = {
        jobid = tostring(game.JobId),
        players = tonumber(#Players:GetPlayers()),
        placeId = tostring(game.PlaceId)
    }

    print("📡 [Tracker] Đang bắn dữ liệu lên API...")
    
    local success, result = pcall(function()
        return KhongKichRequest({
            Url = web_url,
            Method = "POST",
            Headers = { 
                ["Content-Type"] = "application/json"
            },
            Body = HttpService:JSONEncode(data)
        })
    end)

    if success and result then
        if result.StatusCode == 200 or result.StatusCode == 201 then
            print("✅ [Tracker] Gửi thành công! Server này đã lên sóng Web.")
        else
            warn("❌ [Tracker] Web từ chối phản hồi. Mã lỗi: " .. tostring(result.StatusCode))
        end
    else
        warn("🚀 [Tracker] Thất bại! Có thể do Web Render đang ngủ đông, đang thử lại...")
    end
end

-- [HỆ THỐNG QUẢN LÝ SERVER HOP TỰ ĐỘNG - CHẠY SAU KHI QUÉT XONG]
--------------------------------------------------------------------------------
task.spawn(function()
    -- Chờ 12 giây để 16 luồng quét phía trên hoàn tất đợt quét đầu tiên và bắn dữ liệu
    print("⏳ [Server Hop] Đang chờ 12 giây để các luồng quét hoàn tất dữ liệu đợt đầu...")
    task.wait(12)
    
    print("🚀 [Server Hop] Bắt đầu tìm kiếm Server mới để chuyển...")
    local PlaceId = game.PlaceId
    
    -- Thực hiện tìm kiếm server chưa đầy
    local success, servers = pcall(function()
        local url = "https://games.roblox.com/v1/games/"..PlaceId.."/servers/Public?sortOrder=Asc&limit=100"
        local response = game:HttpGet(url)
        return HttpService:JSONDecode(response).data
    end)

    if success and servers then
        local targetServer = nil
        -- Trộn ngẫu nhiên danh sách để tránh lặp lại cùng một server liên tục
        local randomizedServers = {}
        for _, s in pairs(servers) do
            table.insert(randomizedServers, s)
        end
        
        -- Giải thuật xáo trộn Fisher-Yates đơn giản
        for i = #randomizedServers, 2, -1 do
            local j = math.random(i)
            randomizedServers[i], randomizedServers[j] = randomizedServers[j], randomizedServers[i]
        end

        for _, s in pairs(randomizedServers) do
            -- Tìm server còn chỗ trống và không trùng với server hiện tại
            if s.playing and s.maxPlayers and s.playing < s.maxPlayers and s.id ~= game.JobId then
                targetServer = s.id
                break
            end
        end

        if targetServer then
            print("🔗 [Server Hop] Đang kích hoạt cổng Teleport của Blox Fruits tới ID: " .. tostring(targetServer))
            
            -- Chuyển targetServer vào Remote của Server Browser trước khi hop theo yêu cầu của bạn
            pcall(function()
                local serverBrowser = RS:FindFirstChild("__ServerBrowser")
                if serverBrowser and serverBrowser:IsA("RemoteFunction") then
                    serverBrowser:InvokeServer("teleport", targetServer)
                end
            end)
            
            -- Đợi 2 giây xem Remote của game có tự động teleport đi không
            task.wait(2)
            
            -- Fallback: Nếu game chưa chuyển cảnh, dùng TeleportService truyền thống để ép buộc chuyển Server
            print("🌀 [Server Hop] Sử dụng dịch vụ TeleportService dự phòng để kết nối...")
            pcall(function()
                TeleportService:TeleportToPlaceInstance(PlaceId, targetServer, Players.LocalPlayer)
            end)
        else
            warn("⚠️ [Server Hop] Không tìm thấy server nào thích hợp để chuyển!")
            -- Thử lại sau 5 giây nếu không tìm thấy server
            task.wait(5)
            TeleportService:Teleport(PlaceId, Players.LocalPlayer)
        end
    else
        warn("❌ [Server Hop] Thất bại khi lấy danh sách server công khai từ API Roblox!")
        -- Tự động Teleport nhanh sang server ngẫu nhiên bằng cơ chế gốc của Roblox
        task.wait(5)
        pcall(function()
            TeleportService:Teleport(PlaceId, Players.LocalPlayer)
        end)
    end
end)
